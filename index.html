list.map(x => { const empName = x.employees ? `${x.employees.prefix||''}${x.employees.first_name} ${x.employees.last_name}` : 'ไม่พบข้อมูล'; const docTitle = x.document_name || x.doc_title || '-'; const isAdminOrStaff = currentProfile && ['admin','staff'].includes(currentProfile.role); const isMyDoc = currentProfile && currentProfile.employee_id === x.employee_id; return ` ${formatThaiDate(x.created_at)} ${empName} ${x.doc_type||'เอกสารทั่วไป'} ${docTitle} 📥 ดาวน์โหลด / เปิดดู ${(isAdminOrStaff || isMyDoc) ? `` : '-'} `; }).join('') : `ยังไม่มีเอกสารในระบบ`; } async function saveDocument(){ const empId = $('doc_employee_id').value || (currentProfile && currentProfile.employee_id); const docType = $('doc_type').value; const docTitle = $('doc_title').value.trim(); const fileInput = $('doc_file'); $('docFormMsg').textContent = ''; if(!empId || !docTitle || fileInput.files.length === 0){ $('docFormMsg').textContent = 'กรุณากรอกชื่อเอกสารและเลือกไฟล์ที่จะอัปโหลด'; return; } const file = fileInput.files[0]; const fileExt = file.name.split('.').pop(); const fileName = `doc_${Date.now()}_${Math.random().toString(36).substring(2,7)}.${fileExt}`; $('saveDocBtn').textContent = 'กำลังอัปโหลดไฟล์...'; $('saveDocBtn').disabled = true; const {data:uploadData, error:uploadErr} = await supabaseClient.storage.from('documents').upload(fileName, file); if(uploadErr){ $('saveDocBtn').textContent = 'อัปโหลดไฟล์'; $('saveDocBtn').disabled = false; $('docFormMsg').textContent = 'อัปโหลดไฟล์ไม่สำเร็จ (โปรดตรวจสอบการสร้าง Bucket documents): ' + uploadErr.message; return; } const {data:{publicUrl}} = supabaseClient.storage.from('documents').getPublicUrl(fileName); // ส่งบันทึกทั้ง document_name และ doc_title เพื่อป้องกันข้อผิดพลาด const payload = { employee_id: empId, doc_type: docType, document_name: docTitle, file_url: publicUrl, file_name: file.name }; const {error:insertErr} = await supabaseClient.from('documents').insert(payload); $('saveDocBtn').textContent = 'อัปโหลดไฟล์'; $('saveDocBtn').disabled = false; if(insertErr){ $('docFormMsg').textContent = 'บันทึกข้อมูลไม่สำเร็จ: ' + insertErr.message; return; } closeDocumentModal(); await loadDocuments(); }
